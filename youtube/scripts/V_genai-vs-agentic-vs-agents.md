### Demo 3 — AI Agent (Simple function-calling agent in terminal)

**Setup**: A Python script using OpenAI's function calling API. Two tools: `web_search` (simulated) and `calculator`.

**Script: `demo_agent.py`**

```python
from openai import OpenAI
import json

client = OpenAI()

tools = [
    {
        "type": "function",
        "function": {
            "name": "web_search",
            "description": "Search the web for current information",
            "parameters": {
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",
                        "description": "The search query"
                    }
                },
                "required": ["query"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "calculator",
            "description": "Perform a math calculation",
            "parameters": {
                "type": "object",
                "properties": {
                    "expression": {
                        "type": "string",
                        "description": "The math expression to evaluate, e.g. '17900000 * 0.15'"
                    }
                },
                "required": ["expression"]
            }
        }
    }
]

SEARCH_DATA = {
    "population of netherlands": "The population of the Netherlands is approximately 17.9 million as of 2026.",
    "top python web frameworks 2026": "The top Python web frameworks in 2026 are: 1. FastAPI (78k GitHub stars), 2. Django (81k stars), 3. Flask (68k stars)."
}

def web_search(query: str) -> str:
    query_lower = query.lower()
    for key, value in SEARCH_DATA.items():
        if key in query_lower:
            return value
    return f"Search results for '{query}': No specific data found. General information available."

def calculator(expression: str) -> str:
    try:
        result = eval(expression)
        return f"{expression} = {result:,.2f}"
    except Exception as e:
        return f"Error: {e}"

def run_agent(user_question: str):
    print(f"\n{'='*60}")
    print(f"USER: {user_question}")
    print(f"{'='*60}")

    messages = [
        {"role": "system", "content": "You are a helpful assistant. Use the provided tools to answer questions accurately. Always show your reasoning."},
        {"role": "user", "content": user_question}
    ]

    step = 0
    while True:
        step += 1
        print(f"\n--- STEP {step}: THINKING ---")

        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            tools=tools,
            tool_choice="auto"
        )

        message = response.choices[0].message

        if message.tool_calls:
            messages.append(message)
            for tool_call in message.tool_calls:
                fn_name = tool_call.function.name
                fn_args = json.loads(tool_call.function.arguments)

                print(f"ACTION: Calling {fn_name}({fn_args})")

                if fn_name == "web_search":
                    result = web_search(fn_args["query"])
                elif fn_name == "calculator":
                    result = calculator(fn_args["expression"])
                else:
                    result = "Unknown tool"

                print(f"OBSERVE: {result}")

                messages.append({
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "content": result
                })
        else:
            print(f"\n--- FINAL ANSWER ---")
            print(message.content)
            break

if __name__ == "__main__":
    run_agent("What is the population of Netherlands and what's 15% of it?")
```

**Steps on recording day**:
1. Show the code briefly (30 seconds — scroll through, point out the tools list and the think-act-observe loop).
2. Run: `python demo_agent.py`
3. Narrate each step as it prints:
   - "Step 1: It's thinking. It decided to call web_search."

4. Key callout: "See the loop? Think. Act. Observe. Think again. That loop is what makes this an agent, not just a chatbot."
